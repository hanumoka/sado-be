### 5.2 Cornerstone.js 초기화

```typescript
// src/lib/cornerstone/init.ts
import * as cornerstone from '@cornerstonejs/core'
import * as cornerstoneTools from '@cornerstonejs/tools'
import { cornerstoneWADOImageLoader } from '@cornerstonejs/dicom-image-loader'
import dicomParser from 'dicom-parser'

let initialized = false

export async function initCornerstone(): Promise<void> {
  if (initialized) return

  // WebGL 렌더링 엔진 초기화
  await cornerstone.init()

  // DICOM 이미지 로더 설정
  cornerstoneWADOImageLoader.external.cornerstone = cornerstone
  cornerstoneWADOImageLoader.external.dicomParser = dicomParser

  cornerstoneWADOImageLoader.configure({
    useWebWorkers: true,
    decodeConfig: {
      convertFloatPixelDataToInt: false
    }
  })

  // 웹 워커 등록
  cornerstoneWADOImageLoader.webWorkerManager.initialize({
    maxWebWorkers: navigator.hardwareConcurrency || 4,
    startWebWorkersOnDemand: true,
    taskConfiguration: {
      decodeTask: {
        initializeCodecsOnStartup: true
      }
    }
  })

  // 도구 초기화
  cornerstoneTools.init()

  // 도구 등록
  cornerstoneTools.addTool(cornerstoneTools.WindowLevelTool)
  cornerstoneTools.addTool(cornerstoneTools.PanTool)
  cornerstoneTools.addTool(cornerstoneTools.ZoomTool)
  cornerstoneTools.addTool(cornerstoneTools.StackScrollTool)

  initialized = true
}
```

```typescript
// src/features/dicom-viewer/stores/cornerstoneMultiViewerStore.ts
import { create } from 'zustand'
import { subscribeWithSelector, persist } from 'zustand/middleware'

interface ViewerState {
  // 레이아웃
  layout: '1x1' | '1x2' | '2x2' | '2x3' | '3x3'
  activeSlot: number

  // 재생 상태
  playingSlots: Set<number>
  frameIndices: Map<number, number>
  fps: number

  // 캐시
  loadedFrames: Map<string, HTMLImageElement>

  // 액션
  setLayout: (layout: ViewerState['layout']) => void
  setActiveSlot: (slot: number) => void
  togglePlay: (slot: number) => void
  setFrameIndex: (slot: number, index: number) => void
  advanceFrame: (slot: number, totalFrames: number) => void
}

export const useViewerStore = create<ViewerState>()(
  subscribeWithSelector(
    persist(
      (set, get) => ({
        layout: '2x2',
        activeSlot: 0,
        playingSlots: new Set(),
        frameIndices: new Map(),
        fps: 30,
        loadedFrames: new Map(),

        setLayout: (layout) => set({ layout }),

        setActiveSlot: (slot) => set({ activeSlot: slot }),

        togglePlay: (slot) => set((state) => {
          const newPlayingSlots = new Set(state.playingSlots)
          if (newPlayingSlots.has(slot)) {
            newPlayingSlots.delete(slot)
          } else {
            newPlayingSlots.add(slot)
          }
          return { playingSlots: newPlayingSlots }
        }),

        setFrameIndex: (slot, index) => set((state) => {
          const newFrameIndices = new Map(state.frameIndices)
          newFrameIndices.set(slot, index)
          return { frameIndices: newFrameIndices }
        }),

        advanceFrame: (slot, totalFrames) => set((state) => {
          const currentIndex = state.frameIndices.get(slot) || 0
          const nextIndex = (currentIndex + 1) % totalFrames
          const newFrameIndices = new Map(state.frameIndices)
          newFrameIndices.set(slot, nextIndex)
          return { frameIndices: newFrameIndices }
        })
      }),
      {
        name: 'viewer-storage',
        partialize: (state) => ({
          layout: state.layout,
          fps: state.fps
          // loadedFrames는 직렬화하지 않음
        })
      }
    )
  )
)
```

### 6.3 커스텀 Query Hook

```typescript
// src/features/study/hooks/useStudies.ts
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
import { studyService } from '@/lib/services/studyService'
import type { StudyFilter, Study } from '../types'

// 쿼리 키 상수
export const studyKeys = {
  all: ['studies'] as const,
  lists: () => [...studyKeys.all, 'list'] as const,
  list: (filters: StudyFilter) => [...studyKeys.lists(), filters] as const,
  details: () => [...studyKeys.all, 'detail'] as const,
  detail: (id: number) => [...studyKeys.details(), id] as const
}

// Study 목록 조회
export function useStudies(filters: StudyFilter = {}) {
  return useQuery({
    queryKey: studyKeys.list(filters),
    queryFn: () => studyService.getStudies(filters),
    placeholderData: (previousData) => previousData  // 필터 변경 시 이전 데이터 유지
  })
}

// Study 상세 조회
export function useStudy(id: number) {
  return useQuery({
    queryKey: studyKeys.detail(id),
    queryFn: () => studyService.getStudy(id),
    enabled: !!id
  })
}

// Study 삭제
export function useDeleteStudy() {
  const queryClient = useQueryClient()

  return useMutation({
    mutationFn: (id: number) => studyService.deleteStudy(id),
    onSuccess: () => {
      // 목록 캐시 무효화
      queryClient.invalidateQueries({ queryKey: studyKeys.lists() })
    }
  })
}
```

## 7. API 통합

### 7.1 Fetch Wrapper

```typescript
// src/lib/api.ts
const API_BASE_URL = import.meta.env.VITE_API_BASE_URL

export class ApiError extends Error {
  constructor(
    public status: number,
    message: string,
    public code?: string
  ) {
    super(message)
    this.name = 'ApiError'
  }

  static networkError(): ApiError {
    return new ApiError(0, 'Network error')
  }
}

interface RequestOptions extends RequestInit {
  params?: Record<string, string | number | boolean | undefined>
}

async function request<T>(
  endpoint: string,
  options: RequestOptions = {}
): Promise<T> {
  const { params, ...init } = options
  const url = new URL(`${API_BASE_URL}${endpoint}`)

  // 쿼리 파라미터 추가
  if (params) {
    Object.entries(params).forEach(([key, value]) => {
      if (value !== undefined) {
        url.searchParams.append(key, String(value))
      }
    })
  }

  // 기본 헤더
  const headers = new Headers(init.headers)
  if (!headers.has('Content-Type') && !(init.body instanceof FormData)) {
    headers.set('Content-Type', 'application/json')
  }

  // X-Tenant-ID 헤더 (멀티테넌시)
  const tenantId = sessionStorage.getItem('tenantId') || 'default'
  headers.set('X-Tenant-ID', tenantId)

  try {
    const response = await fetch(url.toString(), {
      ...init,
      headers
    })

    // 204 No Content
    if (response.status === 204) {
      return null as T
    }

    // 에러 응답
    if (!response.ok) {
      const errorData = await response.json().catch(() => ({}))
      throw new ApiError(
        response.status,
        errorData.error?.message || response.statusText,
        errorData.error?.code
      )
    }

    return response.json()
  } catch (error) {
    if (error instanceof ApiError) {
      throw error
    }
    throw ApiError.networkError()
  }
}

export const api = {
  get: <T>(endpoint: string, options?: RequestOptions) =>
    request<T>(endpoint, { ...options, method: 'GET' }),

  post: <T>(endpoint: string, body?: unknown, options?: RequestOptions) =>
    request<T>(endpoint, {
      ...options,
      method: 'POST',
      body: body instanceof FormData ? body : JSON.stringify(body)
    }),

  put: <T>(endpoint: string, body?: unknown, options?: RequestOptions) =>
    request<T>(endpoint, {
      ...options,
      method: 'PUT',
      body: JSON.stringify(body)
    }),

  delete: <T>(endpoint: string, options?: RequestOptions) =>
    request<T>(endpoint, { ...options, method: 'DELETE' })
}
```

```typescript
export function CornerstoneSlot({ slotIndex, seriesData }: CornerstoneSlotProps) {
  const viewportRef = useRef<HTMLDivElement>(null)
  const viewportIdRef = useRef<string>(`viewport-${slotIndex}`)

  const { frameIndices, playingSlots, advanceFrame, fps } = useViewerStore()
  const currentFrame = frameIndices.get(slotIndex) || 0
  const isPlaying = playingSlots.has(slotIndex)

  // Viewport 초기화
  useEffect(() => {
    if (!viewportRef.current || !seriesData) return

    const renderingEngine = cornerstone.getRenderingEngine('viewer-engine')
    const viewport = renderingEngine.getViewport(viewportIdRef.current)

    if (!viewport) {
      const element = viewportRef.current
      const viewportInput = {
        viewportId: viewportIdRef.current,
        type: cornerstone.Enums.ViewportType.STACK,
        element,
        defaultOptions: {
          background: [0, 0, 0] as cornerstone.Types.Point3
        }
      }
      renderingEngine.enableElement(viewportInput)
    }

    // 이미지 스택 로드
    const imageIds = seriesData.instances.map(
      (inst) => `wadouri:${API_BASE_URL}/dicomweb/studies/${seriesData.studyUid}/series/${seriesData.seriesUid}/instances/${inst.sopInstanceUid}/rendered`
    )

    const stackViewport = renderingEngine.getViewport(viewportIdRef.current) as cornerstone.Types.IStackViewport
    stackViewport.setStack(imageIds, 0)
    stackViewport.render()

    return () => {
      renderingEngine.disableElement(viewportIdRef.current)
    }
  }, [seriesData])

  // 프레임 변경
  useEffect(() => {
    const renderingEngine = cornerstone.getRenderingEngine('viewer-engine')
    const viewport = renderingEngine?.getViewport(viewportIdRef.current) as cornerstone.Types.IStackViewport

    if (viewport) {
      viewport.setImageIdIndex(currentFrame)
    }
  }, [currentFrame])

  // 재생 애니메이션
  useEffect(() => {
    if (!isPlaying || !seriesData) return

    const interval = 1000 / fps
    const timerId = setInterval(() => {
      advanceFrame(slotIndex, seriesData.instances.length)
    }, interval)

    return () => clearInterval(timerId)
  }, [isPlaying, fps, seriesData, slotIndex, advanceFrame])

  return (
    <div
      ref={viewportRef}
      className="w-full h-full bg-black"
      data-slot={slotIndex}
    />
  )
}
```

```typescript
export function transitionToCornerstone(slot: number) {
  const state = get().slots.get(slot)
  if (state?.mode !== 'mjpeg-playing') return

  set((s) => {
    const newSlots = new Map(s.slots)
    newSlots.set(slot, { ...state, mode: 'transition-prepare' })
    return { slots: newSlots }
  })

  // 루프 경계에서 전환 (200ms 크로스페이드)
  const frameCount = state.mjpegFrames.length
  const currentFrame = state.currentFrame

  // 다음 루프 시작점 대기
  const framesToWait = frameCount - (currentFrame % frameCount)
  const waitTime = framesToWait * (1000 / fps)

  setTimeout(() => {
    set((s) => {
      const newSlots = new Map(s.slots)
      const slotState = newSlots.get(slot)
      if (slotState) {
        newSlots.set(slot, { ...slotState, mode: 'transitioning' })
      }
      return { slots: newSlots }
    })

    // 크로스페이드 완료 후 Cornerstone 활성화
    setTimeout(() => {
      set((s) => {
        const newSlots = new Map(s.slots)
        const slotState = newSlots.get(slot)
        if (slotState) {
          newSlots.set(slot, { ...slotState, mode: 'cornerstone' })
        }
        return { slots: newSlots }
      })
    }, 200)
  }, waitTime)
}
```

```typescript
async function loadFramesProgressive(instanceUid: string, totalFrames: number) {
  const initialFrames = Math.min(20, totalFrames)

  // Phase 1: 초기 프레임 로드 (즉시 재생 가능)

// 2. LRU 캐시 (MinHeap 기반)
class LRUFrameCache {
  private cache = new Map<string, HTMLImageElement>()
  private accessOrder = new MinHeap<string>()
  private maxSize: number

  constructor(maxSizeMB: number) {
    this.maxSize = maxSizeMB * 1024 * 1024
  }

  set(key: string, image: HTMLImageElement) {
    // 캐시 크기 초과 시 가장 오래된 항목 제거
    while (this.getCurrentSize() > this.maxSize) {
      const oldest = this.accessOrder.pop()
      if (oldest) this.cache.delete(oldest)
    }

    this.cache.set(key, image)
    this.accessOrder.push(key, Date.now())
  }

  get(key: string): HTMLImageElement | undefined {
    if (this.cache.has(key)) {
      this.accessOrder.update(key, Date.now())
      return this.cache.get(key)
    }
    return undefined
  }
}

function getResolutionForLayout(layout: string): number {

    default: return 256
  }
}

function advanceAllPlayingFrames() {
  const { playingSlots, frameIndices } = get()

  if (playingSlots.size === 0) return

  playingSlots.forEach((slot) => {
```

## 13. 트러블슈팅

### 13.1 자주 발생하는 문제

#### Cornerstone 초기화 실패

**증상**: `cornerstone is not initialized` 에러

**원인**: 컴포넌트 렌더링 전 초기화 미완료

**해결**:
```typescript
// App.tsx에서 초기화 완료 대기
function App() {
  const [initialized, setInitialized] = useState(false)

  useEffect(() => {
    initCornerstone().then(() => setInitialized(true))
  }, [])

  if (!initialized) return <LoadingScreen />

  return <RouterProvider router={router} />
}
```

#### CORS 에러

**증상**: `Access-Control-Allow-Origin` 에러

**원인**: Backend CORS 설정 누락

**해결**: Nginx 프록시 사용 또는 Backend CORS 설정
```typescript
// vite.config.ts (개발 환경)
export default defineConfig({
  server: {
    proxy: {
      '/api': 'http://localhost:10201',
      '/dicomweb': 'http://localhost:10201'
    }
  }
})
```

#### TanStack Query 캐시 미갱신

**증상**: 데이터 변경 후 목록에 반영 안됨

**원인**: 캐시 무효화 누락

**해결**:
```typescript
const queryClient = useQueryClient()

// 목록 캐시 무효화
queryClient.invalidateQueries({ queryKey: ['studies'] })

// 또는 특정 쿼리만 무효화
queryClient.invalidateQueries({ queryKey: ['studies', 'list'] })
```

#### Zustand persist 직렬화 에러

**증상**: `Failed to serialize` 에러

**원인**: Map, Set, HTMLElement 등 직렬화 불가 타입

**해결**:
```typescript
persist(
  (set) => ({ ... }),
  {
    name: 'storage-key',
    partialize: (state) => ({
      // 직렬화 가능한 필드만 선택
      layout: state.layout,
      fps: state.fps
      // loadedFrames: 제외 (Map<string, HTMLImageElement>)
    })
  }
)
```
